# Intentional absence is part of these switches' interface, not missing configuration.
# Direction alternatives are checked as an exactly-one group by the checker.
RAPT_RV64 # Absent selects RV32; supplied by the build's XLEN flag.
RAPT_NHARTS # Preset-selected cluster hart count; absent selects single-hart.
RAPT_RVFI # Formal instrumentation, enabled only by the formal build.
RAPT_ASSERT_EN # Simulation assertions, supplied by simulation/verification builds.
RAPT_FORMAL_NATIVE # Opt-in native formal lowering of clocked assertion macros.
RAPT_AXI_OBSERVE # Optional simulation AXI observation ports.
RAPT_USE_SRAM_MACRO # Physical SRAM flow; absent selects behavioral storage.
RAPT_SRAM_CTRL_RESET # Preset macro control reset; absent selects the safe zero default.
RAPT_YSYXSOC # Alternate SoC integration selected by its build.
RAPT_PMEM_BYTES # Platform override; rapt_pkg supplies the default PmemBytes value.
RAPT_PMEM_MB # Quote-free platform override in MiB; rapt_pkg derives PmemBytes from it.
RAPT_FETCH_LOOKAHEAD # Preset-selected fetch lookahead; absent disables it.
RAPT_FETCH_WIDE # Preset-selected wide instruction window; absent selects the narrow window.
RAPT_LSU_HUM # Preset-selected hit-under-miss path; absent disables it.
RAPT_L2_EN # Preset-selected L2; absent bypasses it.
RAPT_L2_STORE_WRITEBACK # L2 store policy; absent selects write-through.
RAPT_KEEP_UOP_INST # Synthesis opt-in that keeps expanded instruction words in uops; absent drops them.
RAPT_PC_FULL_WIDTH # Keep XLEN-wide PCs; absent stores RV64 PCs in 40 bits when PA <= 39 bits.
RAPT_NO_FPU # Preset-selected; absent keeps the FEU and F/D in MISA.
RAPT_V_EXTENSION # Preset-selected V; absent removes the integrated VPU.
RAPT_NO_VPU # Explicit scalar override for application presets.
RAPT_ZVBB # Optional Zvbb decoder and execution paths when V is enabled.
RAPT_ZVFHMIN # Optional vector binary16 format conversions when V is enabled.
RAPT_ZVKT # Data-independent timing for the covered vector instructions.
RAPT_ZAWRS # Optional wait-on-reservation-set execution.
RAPT_SSNPM # RV64 U-mode pointer masking through senvcfg.PMM.
RAPT_ZFA # Optional RV64 additional floating-point instructions.
RAPT_ZICCLSM_VECTOR # Split misaligned vector elements in main memory.
RAPT_RVA23U64 # Effective RV64 user profile, derived from the FP and vector configuration.
RAPT_NO_RVA23U64 # Select baseline V and scalar ISA on an RV64 application preset.
RAPT_RVA23S64 # Effective RV64 supervisor profile, derived from RAPT_RVA23U64 unless disabled.
RAPT_NO_RVA23S64 # Keep RVA23U64 without enabling the RVA23S64 supervisor extensions.
RAPT_SVNAPOT # Optional Sv39 / Sv39x4 64 KiB NAPOT leaf page-table entries.
RAPT_SSCOFPMF # Optional scountovf, RV32 mhpmeventh and LCOFI counter-overflow interrupt.
RAPT_SMSTATEEN # Optional Smstateen/Ssstateen state-enable CSRs and access gating.
RAPT_H_EXTENSION # Optional RV64 Hypervisor extension (Sha, Sstc vstimecmp, Svinval H-fences).
RAPT_NO_H_EXTENSION # Keep RVA23S64 sub-extensions without enabling the Hypervisor extension.
RAPT_SMNPM # Optional RV64 HS-mode pointer masking through menvcfg.PMM.
